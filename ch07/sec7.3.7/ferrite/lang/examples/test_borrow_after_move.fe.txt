;; This should fail: cannot borrow after move
(defstruct Point (x i32) (y i32))

(defn consume ((p Point)) i32
  (. p x))

(defn main ()
  (let ((p1 (Point 5 10)))
    (let ((p2 p1))  ;; p1 is moved to p2
      (print (. p2 x))
      (let ((r (borrow p1)))  ;; ERROR: cannot borrow moved value 'p1'
        (print (. r x))))))
